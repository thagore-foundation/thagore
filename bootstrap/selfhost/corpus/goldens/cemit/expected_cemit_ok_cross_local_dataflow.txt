#include <stdint.h>

int32_t helper(void);

int32_t helper(void) {
  return 1;
}

int main(void) {
  int32_t value;
  int32_t backup;
  int8_t active;
  value = helper();
  backup = value;
  active = 1;
  while (active) {
    backup = value;
    active = 0;
    if (active) {
      value = backup;
    }
  }
  return (int)value;
}
