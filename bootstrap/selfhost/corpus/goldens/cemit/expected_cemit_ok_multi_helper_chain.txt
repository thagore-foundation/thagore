#include <stdint.h>

int8_t ready_check(void);
int32_t helper(void);
int32_t bonus(void);

int8_t ready_check(void) {
  return 0;
}

int32_t helper(void) {
  return 1;
}

int32_t bonus(void) {
  return 7;
}

int main(void) {
  int32_t value;
  int8_t active;
  value = helper();
  active = ready_check();
  while (active) {
    value = bonus();
    active = 0;
    if (active) {
      value = helper();
    }
  }
  return (int)value;
}
